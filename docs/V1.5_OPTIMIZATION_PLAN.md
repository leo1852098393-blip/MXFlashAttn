```cpp
// 文件: csrc_native/m2_prefill_kernel.cpp
/**
 * M2 Prefill Kernel: 分块并行 + Shared Memory 优化
 * 
 * 设计思路:
 * 1. 使用 128 个 thread block,每个 block 处理一个 Q 块
 * 2. Shared memory 缓存 K/V 块,减少 global memory 访问
 * 3. 使用 __syncthreads() 同步,保证 online softmax 正确性
 * 4. Warp-level reduce 优化归约操作
 */

#include <cuda_runtime.h>
#include <cuda_fp16.h>

__global__ void m2_prefill_kernel(
    const half* __restrict__ q,     // [batch, seq_q, num_heads, head_dim]
    const half* __restrict__ k,     // [batch, seq_k, num_heads, head_dim]
    const half* __restrict__ v,     // [batch, seq_k, num_heads, head_dim]
    half* __restrict__ out,         // [batch, seq_q, num_heads, head_dim]
    int batch,
    int seq_q,
    int seq_k,
    int num_heads,
    int head_dim,
    int chunk_size                  // 推荐 256
) {
    // Block 维度: [num_heads, seq_q / chunk_size, batch]
    int batch_idx = blockIdx.z;
    int head_idx = blockIdx.y;
    int q_chunk_idx = blockIdx.x;
    
    // Shared memory 分配
    extern __shared__ half smem[];
    half* k_shared = smem;                          // [chunk_size, head_dim]
    half* v_shared = &smem[chunk_size * head_dim];  // [chunk_size, head_dim]
    
    // Thread 局部变量
    int tid = threadIdx.x;
    int q_start = q_chunk_idx * chunk_size;
    int q_end = min(q_start + chunk_size, seq_q);
    
    // Online softmax 状态
    float max_logit = -INFINITY;
    float sum_exp = 0.0f;
    float acc[head_dim];  // 累积输出
    for (int i = 0; i < head_dim; i++) acc[i] = 0.0f;
    
    // 遍历 K/V 块
    int num_k_chunks = (seq_k + chunk_size - 1) / chunk_size;
    
    for (int k_chunk_idx = 0; k_chunk_idx < num_k_chunks; k_chunk_idx++) {
        int k_start = k_chunk_idx * chunk_size;
        int k_end = min(k_start + chunk_size, seq_k);
        int k_chunk_len = k_end - k_start;
        
        // 协作加载 K/V 块到 shared memory
        for (int i = tid; i < k_chunk_len * head_dim; i += blockDim.x) {
            int k_local = i / head_dim;
            int dim = i % head_dim;
            int k_global = k_start + k_local;
            
            int k_offset = ((batch_idx * seq_k + k_global) * num_heads + head_idx) * head_dim + dim;
            k_shared[k_local * head_dim + dim] = k[k_offset];
            v_shared[k_local * head_dim + dim] = v[k_offset];
        }
        __syncthreads();
        
        // 每个 thread 处理一个 Q token
        for (int q_local = tid; q_local < (q_end - q_start); q_local += blockDim.x) {
            int q_global = q_start + q_local;
            
            // 加载 Q 向量
            half q_vec[head_dim];
            int q_offset = ((batch_idx * seq_q + q_global) * num_heads + head_idx) * head_dim;
            for (int d = 0; d < head_dim; d++) {
                q_vec[d] = q[q_offset + d];
            }
            
            // 计算 attention scores
            float scores[chunk_size];
            for (int k_local = 0; k_local < k_chunk_len; k_local++) {
                float dot = 0.0f;
                for (int d = 0; d < head_dim; d++) {
                    dot += __half2float(q_vec[d]) * __half2float(k_shared[k_local * head_dim + d]);
                }
                scores[k_local] = dot / sqrtf((float)head_dim);
            }
            
            // Online softmax 更新
            float chunk_max = -INFINITY;
            for (int k_local = 0; k_local < k_chunk_len; k_local++) {
                if (scores[k_local] > chunk_max) chunk_max = scores[k_local];
            }
            
            float new_max = fmaxf(max_logit, chunk_max);
            float exp_factor = expf(max_logit - new_max);
            
            // 重新缩放旧的累积值
            for (int d = 0; d < head_dim; d++) {
                acc[d] *= exp_factor;
            }
            sum_exp *= exp_factor;
            
            // 累积当前块
            for (int k_local = 0; k_local < k_chunk_len; k_local++) {
                float exp_score = expf(scores[k_local] - new_max);
                sum_exp += exp_score;
                
                for (int d = 0; d < head_dim; d++) {
                    acc[d] += exp_score * __half2float(v_shared[k_local * head_dim + d]);
                }
            }
            
            max_logit = new_max;
        }
        __syncthreads();
    }
    
    // 归一化并写出
    for (int q_local = tid; q_local < (q_end - q_start); q_local += blockDim.x) {
        int q_global = q_start + q_local;
        int out_offset = ((batch_idx * seq_q + q_global) * num_heads + head_idx) * head_dim;
        
        for (int d = 0; d < head_dim; d++) {
            out[out_offset + d] = __float2half(acc[d] / (sum_exp + 1e-10f));
        }
    }
}
```

```python
# 文件: csrc_native/bench_ttft.py
"""
vLLM 流式接口 TTFT 测量
"""
import time
from vllm import LLM, SamplingParams

def measure_ttft_streaming(model_path, prompts, sampling_params):
    """
    使用 vLLM 流式 API 测量 TTFT
    
    Returns:
        ttft_list: 每个 prompt 的首 token 延迟 (秒)
        decode_throughput: decode 阶段 tokens/s
    """
    llm = LLM(model=model_path, max_model_len=4096)
    
    ttft_list = []
    decode_stats = []
    
    for prompt in prompts:
        start_time = time.perf_counter()
        first_token_time = None
        token_count = 0
        
        # 流式生成
        for output in llm.generate(prompt, sampling_params, use_tqdm=False):
            if first_token_time is None:
                first_token_time = time.perf_counter()
                ttft = first_token_time - start_time
                ttft_list.append(ttft)
            
            token_count += 1
        
        end_time = time.perf_counter()
        decode_time = end_time - first_token_time
        decode_tps = token_count / max(decode_time, 1e-6)
        decode_stats.append(decode_tps)
    
    return {
        "ttft_mean": np.mean(ttft_list),
        "ttft_p50": np.percentile(ttft_list, 50),
        "ttft_p95": np.percentile(ttft_list, 95),
        "ttft_p99": np.percentile(ttft_list, 99),
        "decode_tps_mean": np.mean(decode_stats),
    }

# 使用示例
if __name__ == "__main__":
    results = measure_ttft_streaming(
        model_path="/root/models/Qwen3-4B",
        prompts=["你好" * i for i in [10, 50, 100, 500, 1000]],
        sampling_params=SamplingParams(max_tokens=100)
    )
    print(f"TTFT P50: {results['ttft_p50']:.3f}s")
    print(f"TTFT P95: {results['ttft_p95']:.3f}s")
```

#### 优化方案: 基于历史统计的自适应调度

```python
# 文件: mxflashattn/adaptive_policy.py
"""
自适应调度策略: 根据历史性能动态选择后端
"""
import json
from pathlib import Path
from collections import defaultdict

class AdaptiveDispatcher:
    """
    基于历史统计的自适应调度器
    
    核心思路:
    1. 记录每种 (batch, seqlen, head_dim, GQ) 组合的性能
    2. 动态更新最优后端选择
    3. 对新组合使用保守策略(vendor/v7)
    4. 定期导出性能 profile,用于下次启动
    """
    
    def __init__(self, profile_path=None, cold_start_policy="vendor"):
        self.profile_path = profile_path or Path.home() / ".mxflashattn_profile.json"
        self.cold_start_policy = cold_start_policy
        
        # 历史性能统计: {(batch, seqlen, head_dim, gq): {"m1": 1230.5, "v7": 450.2, "vendor": 1200.0}}
        self.perf_stats = defaultdict(dict)
        self.load_profile()
    
    def load_profile(self):
        """从磁盘加载历史 profile"""
        if self.profile_path.exists():
            with open(self.profile_path) as f:
                data = json.load(f)
                self.perf_stats = {tuple(json.loads(k)): v for k, v in data.items()}
    
    def save_profile(self):
        """保存 profile 到磁盘"""
        data = {json.dumps(k): v for k, v in self.perf_stats.items()}
        with open(self.profile_path, 'w') as f:
            json.dump(data, f, indent=2)
    
    def select_backend(self, batch, seqlen, head_dim, gq, is_prefill=False):
        """
        选择最优后端
        
        决策逻辑:
        1. 如果是 prefill: 使用 m2_prefill(如果可用) 或 vendor
        2. 查询历史统计,选择最快的后端
        3. 如果没有历史数据: 使用 cold_start_policy
        """
        if is_prefill:
            # Prefill 优先 M2 (如果已实现且快于 vendor)
            # 否则用 vendor
            return "m2_prefill" if self.has_m2_prefill() else "vendor"
        
        # Decode 场景
        key = (batch, seqlen, head_dim, gq)
        
        if key in self.perf_stats:
            # 有历史数据: 选择最快的
            stats = self.perf_stats[key]
            best_backend = max(stats, key=stats.get)
            return best_backend
        else:
            # 冷启动: 使用保守策略
            # 大 batch 长序列 -> M1
            # 小 batch 短序列 -> vendor
            if batch >= 16 and seqlen >= 512:
                return "m1_split_kv"
            elif batch >= 8 and seqlen >= 256:
                return "v7_native"  # 试探性使用 v7
            else:
                return "vendor"  # 保守策略
    
    def record_performance(self, batch, seqlen, head_dim, gq, backend, throughput):
        """记录一次执行的性能"""
        key = (batch, seqlen, head_dim, gq)
        self.perf_stats[key][backend] = throughput
    
    def has_m2_prefill(self):
        """检查 M2 prefill kernel 是否可用"""
        try:
            from mxflashattn.kernels import m2_prefill
            return True
        except ImportError:
            return False

# 集成到 API
# 文件: mxflashattn/api.py
_adaptive_dispatcher = AdaptiveDispatcher()

def flash_attn_func(q, k, v, causal=False):
    """MXFlashAttn 主 API - 自适应调度版本"""
    batch, seqlen, num_heads, head_dim = q.shape
    gq = num_heads // k.shape[2]
    
    # 自适应选择后端
    backend = _adaptive_dispatcher.select_backend(
        batch=batch,
        seqlen=seqlen,
        head_dim=head_dim,
        gq=gq,
        is_prefill=(q.shape[1] > 1)  # 简化判断
    )
    
    # 执行并记录性能
    import time
    start = time.perf_counter()
    
    if backend == "m1_split_kv":
        out = m1_split_kv_decode(q, k, v)
    elif backend == "v7_native":
        out = v7_native_decode(q, k, v)
    elif backend == "vendor":
        out = vendor_flash_attn(q, k, v)
    else:
        raise ValueError(f"Unknown backend: {backend}")
    
    elapsed = time.perf_counter() - start
    throughput = (batch * seqlen) / elapsed
    
    # 记录性能(异步,不阻塞主流程)
    _adaptive_dispatcher.record_performance(batch, seqlen, head_dim, gq, backend, throughput)
    
    return out
```

#### 目标

**阶段1: KV Cache 量化 (影响小,易实现)**

```python
# 文件: mxflashattn/quantization/fp8_kv.py
"""
KV Cache FP8 量化
"""
import torch

def quantize_kv_fp8(k, v):
    """
    将 KV cache 量化到 FP8E4M3
    
    Args:
        k, v: [batch, seqlen, num_kv_heads, head_dim] FP16
    
    Returns:
        k_fp8, v_fp8: FP8 张量
        scale_k, scale_v: 反量化因子
    """
    # Per-tensor 量化(简单版本)
    k_max = torch.abs(k).max()
    v_max = torch.abs(v).max()
    
    scale_k = k_max / 448.0  # FP8E4M3 最大值
    scale_v = v_max / 448.0
    
    k_fp8 = (k / scale_k).to(torch.float8_e4m3fn)
    v_fp8 = (v / scale_v).to(torch.float8_e4m3fn)
    
    return k_fp8, v_fp8, scale_k, scale_v

def dequantize_kv_fp8(k_fp8, v_fp8, scale_k, scale_v):
    """反量化"""
    k = k_fp8.to(torch.float16) * scale_k
    v = v_fp8.to(torch.float16) * scale_v
    return k, v

# 集成到 M1 kernel
def m1_split_kv_decode_fp8(q, k_fp8, v_fp8, scale_k, scale_v):
    """支持 FP8 KV cache 的 M1 decode"""
    # 选项1: 先反量化再计算(简单,性能一般)
    k = dequantize_kv_fp8(k_fp8, scale_k)
    v = dequantize_kv_fp8(v_fp8, scale_v)
```

### 6. 多卡 Tensor Parallelism 🟢

#### 目标
- 2 卡: 吞吐 ≥ 1.7x 单卡
- 4 卡: 吞吐 ≥ 3.2x 单卡
- 支持 Qwen-14B/72B 等大模型

#### 实现方案

```python
# 文件: mxflashattn/distributed/tensor_parallel.py
"""
Tensor Parallelism: 按 attention heads 切分
"""
import torch
import torch.distributed as dist

def flash_attn_tp(q, k, v, world_size=2, rank=0):
    """
    Tensor Parallel 版本的 FlashAttention
    
    原理:
    - 每个 GPU 负责部分 attention heads
    - 本地计算 attention
    - All-Reduce 同步结果
    
    通信量: O(batch * seqlen * hidden_dim) / world_size
    """
    num_heads = q.shape[2]
    heads_per_gpu = num_heads // world_size
    
    # 切分 heads
    head_start = rank * heads_per_gpu
    head_end = (rank + 1) * heads_per_gpu
    
    q_local = q[:, :, head_start:head_end, :]
    k_local = k[:, :, head_start:head_end, :]
    v_local = v[:, :, head_start:head_end, :]
    
    # 本地 attention (使用 M1 kernel)
    out_local = flash_attn_func(q_local, k_local, v_local)
    
    # All-Reduce 同步
    if world_size > 1:
        dist.all_reduce(out_local, op=dist.ReduceOp.SUM)
    
    return out_local

# vLLM 集成
# 文件: integrations/vllm/backend_tp.py
class MXFlashAttnBackendTP(MXFlashAttnBackend):
    """支持 TP 的 vLLM 后端"""
    
    def forward_decode(self, query, key_cache, value_cache, block_tables):
        # 自动检测 TP 环境
        if torch.distributed.is_initialized():
            world_size = torch.distributed.get_world_size()
            rank = torch.distributed.get_rank()
            return flash_attn_tp(query, key_cache, value_cache, world_size, rank)
        else:
            # 单卡路径
            return super().forward_decode(query, key_cache, value_cache, block_tables)
```
